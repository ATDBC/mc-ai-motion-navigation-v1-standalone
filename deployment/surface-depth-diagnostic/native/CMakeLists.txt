cmake_minimum_required(VERSION 3.20)
project(surface_cache_diagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED)
set(CLIPPER "${CMAKE_CURRENT_SOURCE_DIR}/vendor/clipper2")
add_library(surface_cache SHARED cache.cpp jni.cpp "${CLIPPER}/src/clipper.engine.cpp")
target_include_directories(surface_cache PRIVATE "${CLIPPER}/include" ${JNI_INCLUDE_DIRS})
if(MSVC)
  target_compile_options(surface_cache PRIVATE /O2 /EHsc /W4)
else()
  target_compile_options(surface_cache PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
